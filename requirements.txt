# desisim's abstract dependencies are declared in setup.cfg (install_requires),
# so they are deliberately NOT repeated here; install them with
#
#     pip install .
#
# This file lists only what install_requires cannot express: packages that are
# not installable from PyPI, and sibling DESI packages that we deliberately test
# against their development branch rather than their released version.
#
# desiutil is intentionally absent: it is needed at build time by several of the
# installs below, so the GitHub Actions workflow installs it explicitly, pinned
# to a known-good version, in the same pip invocation as this file.

# specter is needed at runtime, but only indirectly, so don't remove this line:
# desisim's pixsim path operates on a specter PSF object (see pixsim.simulate)
# obtained from desimodel.io.load_psf(), and desimodel imports specter.psf
# itself without declaring specter in its own install_requires.  This line is
# therefore the only thing that puts specter in the environment; without it
# py/desisim/test/test_pixsim.py fails in load_psf with an ImportError.
#
# desisim's own single `import specter` (py/desisim/io.py) is just a version
# stamp inside a try/except ImportError, and is NOT the reason this is here.
#
# It cannot move to install_requires: the `specter` name on PyPI belongs to an
# unrelated BDD testing library, while DESI's specter declares that same name.
git+https://github.com/desihub/specter.git@main#egg=specter

# Sibling DESI packages.  install_requires names these without a version so
# that an ordinary `pip install desisim` gets their PyPI releases; here we
# override that to test against main.
git+https://github.com/desihub/desimodel.git@main#egg=desimodel
# Don't forget to install desimodel test data.
git+https://github.com/desihub/desispec.git@main#egg=desispec
git+https://github.com/desihub/desitarget.git@main#egg=desitarget
# specsim 1.0.0 is on PyPI and is covered by install_requires; main is used
# here only to stay ahead of the last tag.  Drop this line to test against the
# release instead.
git+https://github.com/desihub/specsim.git@main#egg=specsim

# simqso is not on PyPI, and its install script requires numpy, so the workflow
# installs it in a separate step once numpy is present.
# git+https://github.com/imcgreer/simqso.git@v1.2.4#egg=simqso
